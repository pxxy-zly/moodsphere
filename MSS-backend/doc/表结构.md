# 心境气象站(moodsphere)数据库表结构 - 表名+字段+类型版
# 数据库：MySQL 8.x | 字符集：utf8mb4

========================================
一、基础配置模块
========================================
1. biz_user_info（用户扩展信息表）
   id: bigint (主键)
   user_id: bigint
   nick_name: varchar(50)
   avatar: varchar(500)
   gender: tinyint
   birthday: date
   signature: varchar(200)
   emotion_level: int
   record_days: int
   total_records: int
   anonymous_projection: tinyint
   last_record_time: datetime
   del_flag: tinyint

2. biz_user_auth（第三方登录绑定表）
   id: bigint (主键)
   user_id: bigint
   auth_type: varchar(30)
   openid: varchar(100)
   unionid: varchar(100)
   session_key: varchar(255)
   auth_status: tinyint
   last_login_time: datetime
   last_login_ip: varchar(64)
   del_flag: tinyint

3. biz_tag（情绪标签表）
   id: bigint (主键)
   tag_name: varchar(50)
   tag_code: varchar(50)
   tag_type: varchar(20)
   sort_order: int
   status: tinyint
   del_flag: tinyint

4. biz_scene（场景标签表）
   id: bigint (主键)
   scene_name: varchar(50)
   scene_code: varchar(50)
   icon: varchar(200)
   sort_order: int
   status: tinyint
   del_flag: tinyint

========================================
二、情绪记录模块
========================================
5. biz_mood_record（情绪记录主表）
   id: bigint (主键)
   user_id: bigint
   source_type: tinyint
   record_type: tinyint
   content_text: text
   voice_duration: int
   emotion_intensity: tinyint
   record_time: datetime
   record_status: tinyint
   analyze_status: tinyint
   is_public: tinyint
   province: varchar(50)
   city: varchar(50)
   district: varchar(50)
   longitude: decimal(10,6)
   latitude: decimal(10,6)
   risk_level: tinyint
   del_flag: tinyint

6. biz_record_tag_rel（情绪记录与标签关联表）
   id: bigint (主键)
   record_id: bigint
   tag_id: bigint
   source: tinyint

7. biz_record_scene_rel（情绪记录与场景关联表）
   id: bigint (主键)
   record_id: bigint
   scene_id: bigint
   source: tinyint

8. biz_mood_asset（情绪素材表）
   id: bigint (主键)
   record_id: bigint
   asset_type: tinyint
   file_url: varchar(500)
   file_size: bigint
   mime_type: varchar(100)
   duration: int
   thumbnail_url: varchar(500)
   width: int
   height: int
   status: tinyint
   del_flag: tinyint

========================================
三、AI分析模块
========================================
9. biz_ai_analysis_result（AI分析结果表）
   id: bigint (主键)
   record_id: bigint
   primary_emotion: varchar(50)
   secondary_emotion: varchar(50)
   emotion_keywords: varchar(500)
   emotion_scores: text
   scene_recognition: varchar(255)
   risk_level: tinyint
   ai_summary: varchar(500)
   provider: varchar(50)
   model_name: varchar(100)
   status: tinyint
   del_flag: tinyint

10. biz_emotion_vector（情绪向量表）
    id: bigint (主键)
    record_id: bigint
    valence: decimal(6,4)
    arousal: decimal(6,4)
    anxiety: decimal(6,4)
    calmness: decimal(6,4)
    loneliness: decimal(6,4)
    fatigue: decimal(6,4)
    anger: decimal(6,4)
    hope: decimal(6,4)
    confidence: decimal(6,4)
    confidence_score: decimal(6,4)
    del_flag: tinyint

========================================
四、天气映射模块
========================================
11. biz_weather_mapping（情绪天气映射结果表）
    id: bigint (主键)
    record_id: bigint
    weather_code: varchar(20)
    weather_name: varchar(50)
    sky_type: tinyint
    cloud_density: tinyint
    rain_intensity: tinyint
    lightning_intensity: tinyint
    wind_speed: tinyint
    fog_intensity: tinyint
    color_temperature: int
    saturation: tinyint
    particle_style: varchar(20)
    del_flag: tinyint

12. biz_weather_snapshot（天气快照表）
    id: bigint (主键)
    user_id: bigint
    snapshot_date: date
    weather_code: varchar(20)
    weather_name: varchar(50)
    sky_type: tinyint
    cloud_density: tinyint
    rain_intensity: tinyint
    lightning_intensity: tinyint
    wind_speed: tinyint
    fog_intensity: tinyint
    color_temperature: int
    saturation: tinyint
    particle_style: varchar(20)
    record_count: int
    del_flag: tinyint

========================================
五、报告模块
========================================
13. biz_mood_report（情绪报告表）
    id: bigint (主键)
    user_id: bigint
    report_type: tinyint
    period_start: date
    period_end: date
    title: varchar(100)
    summary: varchar(500)
    dominant_emotion: varchar(50)
    emotion_trend: text
    statistic_json: text
    pdf_url: varchar(500)
    status: tinyint
    del_flag: tinyint

14. biz_report_detail（报告详情表）
    id: bigint (主键)
    report_id: bigint
    section_type: varchar(50)
    content_json: longtext
    sort_order: int
    del_flag: tinyint

========================================
六、星球投影模块
========================================
15. biz_globe_projection（星球投影记录表）
    id: bigint (主键)
    record_id: bigint
    user_id: bigint
    projection_time: datetime
    province: varchar(50)
    city: varchar(50)
    longitude: decimal(10,6)
    latitude: decimal(10,6)
    emotion_type: varchar(20)
    intensity: tinyint
    anonymous_id: varchar(64)
    status: tinyint
    del_flag: tinyint

16. biz_region_emotion_stats（区域情绪统计表）
    id: bigint (主键)
    region_code: varchar(20)
    region_name: varchar(50)
    level: tinyint
    stat_date: date
    stat_hour: tinyint
    total_records: int
    dominant_emotion: varchar(20)
    emotion_distribution: text
    avg_intensity: decimal(6,4)
    active_user_count: int

========================================
七、风险预警模块
========================================
17. biz_risk_warning（风险预警表）
    id: bigint (主键)
    record_id: bigint
    user_id: bigint
    risk_level: tinyint
    risk_type: varchar(50)
    risk_reason: varchar(255)
    warning_content: text
    status: tinyint
    notify_status: tinyint
    handle_result: varchar(255)
    del_flag: tinyint

========================================
八、通知模块
========================================
18. biz_notification（消息通知表）
    id: bigint (主键)
    user_id: bigint
    type: tinyint
    title: varchar(100)
    content: varchar(500)
    data: text
    biz_id: bigint
    jump_url: varchar(500)
    is_read: tinyint
    push_time: datetime
    expire_time: datetime
    del_flag: tinyint

========================================
九、用户配置模块
========================================
19. biz_user_config（用户配置表）
    id: bigint (主键)
    user_id: bigint
    anonymous_projection: tinyint
    ai_personalized_analysis: tinyint
    daily_remind_enabled: tinyint
    daily_remind_time: varchar(10)
    weekly_report_enabled: tinyint
    language: varchar(20)
    theme_mode: varchar(20)
    del_flag: tinyint

========================================
初始化数据（核心值）
========================================
# 情绪标签初始化（biz_tag）
tag_name(varchar(50)): 开心、焦虑、疲惫、平静、委屈、期待、孤独、烦躁、悲伤、温暖、迷茫、充满希望
tag_code(varchar(50)): happy、anxious、tired、calm、wronged、expect、lonely、irritable、sad、warm、confused、hopeful

# 场景标签初始化（biz_scene）
scene_name(varchar(50)): 工作、学习、家庭、社交、感情、睡眠、健康、娱乐、运动、独处
scene_code(varchar(50)): work、study、family、social、love、sleep、health、entertainment、sport、alone
